// ============================================
// SCRIPTS AND COMMANDS REFERENCE
// ============================================

// ---- Backend Tests ----

// Run backend tests
cd backend
./gradlew test --no-daemon

// Run backend tests with coverage
cd backend
./gradlew test jacocoTestReport --no-daemon


// ---- Frontend Tests ----

// Run frontend tests
cd frontend
npm test

// Run frontend tests with coverage
cd frontend
npm run test:coverage


// ---- SonarQube Scans ----

// Backend scan (requires SONAR_TOKEN)
cd backend
./gradlew sonar --no-daemon

// Frontend scan (requires SONAR_TOKEN)
cd frontend
npm run sonar

// Both scans (sequential)
// PowerShell:
.\run-sonar-both.ps1


// ---- Builds ----

// Build backend (skip tests)
cd backend
./gradlew clean build -x test

// Build frontend
cd frontend
npm ci
npm run build


// ============================================
// MAIN DOCKER SERVICES (docker-compose.yml)
// ============================================

// Start main services
// Backend:  http://localhost:8080
// Frontend: http://localhost:5173
docker-compose up -d

// Stop main services
docker-compose down

// Rebuild and start services
docker-compose up -d --build

// Restart services
docker-compose restart

// Stop services (keep containers)
docker-compose stop

// Start existing containers
docker-compose start


// ============================================
// TEST DATABASE (docker-compose.test.yml)
// ============================================

// Start test DB
docker-compose -f docker-compose.test.yml up -d

// Stop test DB
docker-compose -f docker-compose.test.yml down

// ============================================
// SONARQUBE (docker-compose.sonar.yml)
// ============================================

// Start SonarQube
// http://localhost:9000
docker-compose -f docker-compose.sonar.yml up -d

// Stop SonarQube
docker-compose -f docker-compose.sonar.yml down

// ============================================
// ENVIRONMENT VARIABLES
// ============================================

// Set Sonar token
$env:SONAR_TOKEN = 'token'

// .env for main services:
//   SPRING_DATASOURCE_URL
//   SPRING_DATASOURCE_USERNAME
//   SPRING_DATASOURCE_PASSWORD
//   SPRING_JPA_HIBERNATE_DDL_AUTO
//   SPRING_JPA_SHOW_SQL
//   SPRING_FLYWAY_ENABLED
//   VITE_MAPBOX_TOKEN

// WEB INTERFACES

http://localhost:9000

http://localhost:8080

http://localhost:5173

backend/build/reports/jacoco/test/html/index.html

frontend/coverage/lcov-report/index.html

// CI/CD PIPELINE STAGES

// Build Stage:
//   - backend-build
//   - frontend-build

// Test Stage:
//   - backend-test
//   - frontend-test

//   - sonarqube-backend (allow failure)

// Deploy Stage:
//   - deploy-backend

// NOTES
